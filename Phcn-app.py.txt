from datetime import datetime

# --- CƠ SỞ DỮ LIỆU SQLITE ---
DB_NAME = "phcn_clinical.db"


def init_db():
  conn = sqlite3.connect(DB_NAME)
  c = conn.cursor()
  # Bảng Bệnh nhân
  c.execute("""
        CREATE TABLE IF NOT EXISTS patients (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            code TEXT UNIQUE,
            name TEXT,
            age INTEGER,
            gender TEXT,
            phone TEXT,
            address TEXT,
            created_at TEXT
        )
    """)
  # Bảng Lượng giá & Hồ sơ khám (có lưu kết quả phân tích từ Gemini)
  c.execute("""
        CREATE TABLE IF NOT EXISTS clinical_records (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            patient_id INTEGER,
            exam_date TEXT,
            region TEXT,
            symptoms TEXT,
            red_flags TEXT,
            test_results TEXT,
            gemini_analysis TEXT,
            plan TEXT,
            FOREIGN KEY (patient_id) REFERENCES patients (id)
        )
    """)
  conn.commit()
  conn.close()


init_db()

st.set_page_config(
    page_title="Trợ lý lâm sàng PHCN kết nối Gemini", page_icon="🩺", layout="wide"

# --- SIDEBAR: CẤU HÌNH API KEY ---

api_key_input = st.sidebar.text_input(
    "Google Gemini API Key", type="password", help="Nhập API Key của bạn để sử dụng AI"
)

menu = st.sidebar.selectbox(
    "Chọn chức năng",
    [
        "Quản lý Bệnh nhân",
        "Khám & Lượng giá Lâm sàng",
        "Theo dõi Tiến triển",
        "Sao lưu / Khôi phục Dữ liệu",
    ],
)

# --- CHỨC NĂNG 1: QUẢN LÝ BỆNH NHÂN ---
if menu == "Quản lý Bệnh nhân":
  st.header("👤 Quản lý Hồ sơ Bệnh nhân")

  with st.form("patient_form"):
    col1, col2 = st.columns(2)
    with col1:
      name = st.text_input("Họ và tên bệnh nhân")
      age = st.number_input("Tuổi", min_value=1, max_value=120, value=30)
      gender = st.selectbox("Giới tính", ["Nam", "Nữ", "Khác"])
    with col2:
      phone = st.text_input("Số điện thoại")
      address = st.text_input("Địa chỉ")

    submitted = st.form_submit_button("Tạo hồ sơ mới")
    if submitted:
      if name:
        code = f"BN{datetime.now().strftime('%Y%m%d%H%M%S')}"
        created_at = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        conn = sqlite3.connect(DB_NAME)
        c = conn.cursor()
        c.execute(
            "INSERT INTO patients (code, name, age, gender, phone, address,"
            " created_at) VALUES (?, ?, ?, ?, ?, ?, ?)",
            (code, name, age, gender, phone, address, created_at),
        )
        conn.commit()
        conn.close()
        st.success(
            f"Đã tạo thành công hồ sơ cho bệnh nhân: **{name}** (Mã: {code})"
        )
      else:
        st.warning("Vui lòng nhập tên bệnh nhân!")

  st.markdown("---")
  st.subheader("Danh sách bệnh nhân đã lưu")
  conn = sqlite3.connect(DB_NAME)
  c = conn.cursor()
  c.execute("SELECT code, name, age, gender, phone, created_at FROM patients")
  rows = c.fetchall()
  conn.close()

  if rows:
    for r in rows:
      st.write(
          f"🔹 **Mã:** {r[0]} | **Họ tên:** {r[1]} | **Tuổi:** {r[2]} | **Giới"
          f" tính:** {r[3]} | **SĐT:** {r[4]} | **Ngày tạo:** {r[5]}"
      )
  else:
    st.info("Chưa có bệnh nhân nào trong cơ sở dữ liệu.")

# --- CHỨC NĂNG 2: KHÁM & LƯỢNG GIÁ LÂM SÀNG ---
elif menu == "Khám & Lượng giá Lâm sàng":
  st.header("📋 Khám, Sàng lọc & Phân tích Lâm sàng cùng Gemini")

  conn = sqlite3.connect(DB_NAME)
  c = conn.cursor()
  c.execute("SELECT id, code, name FROM patients")
  patients = c.fetchall()
  conn.close()

  if not patients:
    st.warning("Vui lòng tạo hồ sơ bệnh nhân trước khi tiến hành khám.")
  else:
    patient_dict = {f"{p[1]} - {p[2]}": p[0] for p in patients}
    selected_p = st.selectbox("Chọn bệnh nhân", list(patient_dict.keys()))
    patient_id = patient_dict[selected_p]

    region = st.selectbox(
        "Vùng cơ thể cần lượng giá",
        ["Cột sống thắt lưng", "Cột sống cổ", "Vai", "Khớp gối"],
    )

    st.markdown("### 1. Khai thác triệu chứng")
    symptoms = st.text_area(
        "Mô tả triệu chứng (Vị trí đau, tính chất, thời gian, hướng lan...)"
    )

    st.markdown("### 2. Sàng lọc dấu hiệu cờ đỏ (Red Flags)")
    red_flags_list = RED_FLAGS_DATA.get(region, [f"Dấu hiệu bất thường tại {region}"])
    selected_red_flags = []
    for rf in red_flags_list:
      if st.checkbox(rf, key=rf):
        selected_red_flags.append(rf)

    if selected_red_flags:
      st.error(
          "⚠️ **CẢNH BÁO CỜ ĐỎ:** Phát hiện dấu hiệu nguy hiểm! Khuyến cáo chuyển"
          " khám chuyên khoa."
      )

    st.markdown("### 3. Đề xuất nghiệm pháp lâm sàng")
    suggested_tests = TESTS_DATA.get(region, [])
    test_results_dict = {}
    if suggested_tests:
      for test in suggested_tests:
        with st.expander(f"🧪 {test['name']}"):
          st.write(f"**Mục đích:** {test['purpose']}")
          st.write(f"**Cách thực hiện:** {test['instruction']}")
          st.write(f"**Tiêu chuẩn dương tính:** {test['positive']}")
          res = st.radio(
              f"Kết quả cho {test['name']}",
              ["Chưa thực hiện", "Âm tính (-)", "Dương tính (+)"],
              key=test["name"],
          )
          test_results_dict[test["name"]] = res
    else:
      st.info("Đang cập nhật thêm nghiệm pháp cho vùng này.")

    st.markdown("### 4. Phân tích & Gợi ý từ Gemini AI")
    gemini_analysis_result = ""

    if st.button("🤖 Yêu cầu Gemini Phân tích Lâm sàng"):
      if not api_key_input:
        st.error(
            "Vui lòng nhập Google Gemini API Key ở thanh bên trái (Sidebar) để"
            " sử dụng tính năng này."
        )
      elif not symptoms:
        st.warning("Vui lòng nhập mô tả triệu chứng trước khi yêu cầu AI phân tích.")
      else:
        with st.spinner("Gemini đang phân tích dữ liệu lâm sàng..."):
          try:
            client = genai.Client(api_key=api_key_input)
            prompt = f"""
                        Bạn là một chuyên gia cao cấp về Vật lý trị liệu và Phục hồi chức năng (PHCN).
                        Hãy phân tích thông tin ca lâm sàng sau đây và đưa ra nhận định chuyên môn:
                        - Vùng cơ thể: {region}
                        - Triệu chứng bệnh nhân: {symptoms}
                        - Các cờ đỏ (Red flags) đã chọn: {', '.join(selected_red_flags) if selected_red_flags else 'Không có'}
                        - Kết quả nghiệm pháp thực hiện: {str(test_results_dict)}

                        Hãy đưa ra:
                        1. Nhận định sơ bộ về nguyên nhân cơ học / tổn thương cấu trúc.
                        2. Đánh giá mức độ rủi ro dựa trên cờ đỏ và kết quả nghiệm pháp.
                        3. Gợi ý hướng can thiệp / bài tập PHCN ban đầu phù hợp và an toàn.
                        """
            
            response = client.models.generate_content(
                model="gemini-2.5-flash",
                contents=prompt,
            )
            gemini_analysis_result = response.text
            st.success("Phân tích hoàn tất!")
          except Exception as e:
            st.error(f"Đã xảy ra lỗi khi kết nối với Gemini API: {e}")

    # Hiển thị kết quả phân tích (nếu có)
    if gemini_analysis_result:
      st.markdown("#### Kết quả phân tích từ AI:")
      st.markdown(gemini_analysis_result)

    plan = st.text_area("Kế hoạch can thiệp PHCN thực tế (có thể chỉnh sửa dựa trên gợi ý AI)")

    if st.button("Lưu hồ sơ khám"):
      exam_date = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
      rf_str = ", ".join(selected_red_flags)
      tr_str = str(test_results_dict)

      conn = sqlite3.connect(DB_NAME)
      c = conn.cursor()
      c.execute(
          """
                INSERT INTO clinical_records (patient_id, exam_date, region, symptoms, red_flags, test_results, gemini_analysis, plan)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?)
            """,
          (
              patient_id,
              exam_date,
              region,
              symptoms,
              rf_str,
              tr_str,
              gemini_analysis_result,
              plan,
          ),
      )
      conn.commit()
      conn.close()
      st.success("Đã lưu kết quả khám và lượng giá thành công vào cơ sở dữ liệu!")

# --- CHỨC NĂNG 3: THEO DÕI TIẾN TRIỂN ---
elif menu == "Theo dõi Tiến triển":
  st.header("📈 Theo dõi Tiến triển Bệnh nhân")

  conn = sqlite3.connect(DB_NAME)
  c = conn.cursor()
  c.execute("SELECT id, code, name FROM patients")
  patients = c.fetchall()
  conn.close()

  if not patients:
    st.info("Chưa có dữ liệu bệnh nhân.")
  else:
    patient_dict = {f"{p[1]} - {p[2]}": p[0] for p in patients}
    selected_p = st.selectbox(
        "Chọn bệnh nhân xem lịch sử", list(patient_dict.keys())
    )
    patient_id = patient_dict[selected_p]

    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute(
        """
            SELECT exam_date, region, symptoms, red_flags, gemini_analysis, plan
            FROM clinical_records WHERE patient_id = ?
        """,
        (patient_id,),
    )
    records = c.fetchall()
    conn.close()

    if records:
      for idx, rec in enumerate(records, 1):
        st.markdown(f"### Lần khám #{idx} - Ngày: {rec[0]}")
        st.write(f"**Vùng lượng giá:** {rec[1]}")
        st.write(f"**Triệu chứng:** {rec[2]}")
        st.write(f"**Cờ đỏ ghi nhận:** {rec[3] if rec[3] else 'Không có'}")
        if rec[4]:
          st.markdown(f"**Phân tích từ AI:**\n{rec[4]}")
        st.write(f"**Kế hoạch can thiệp:** {rec[5]}")
        st.markdown("---")
    else:
      st.info("Bệnh nhân này chưa có bản ghi lượng giá nào.")
